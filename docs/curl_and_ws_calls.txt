#### Sample RestAPI calls

## Keyer
  # Remove a keyer together with its ports. Refused with 409 while the keyer is plugged in.
curl -X DELETE http://localhost:5052/api/v1/config/devices/SM_DEMO_SM_1

  # Name a keyer. Up to 32 characters without surrounding spaces, refused with 400 otherwise.
  # Refused with 409 if another keyer has this name or serial number, ignoring case.
  # "" or null go back to the default name, the serial number.
curl -X PATCH http://localhost:5052/api/v1/config/devices/SM_DEMO_SM_1 \
  -H 'Content-Type: application/json' \
  -d '{"name": "Shack"}'

## SM / SMD
  # Add an item to Antenna List


curl -X PATCH http://localhost:5052/api/v1/config/devices/SM_DEMO_SM_1 \
  -H 'Content-Type: application/json' \
  -d '{
    "serial": "SM_DEMO_SM_1",
    "sm": {
      "obj": {
        "action": "add",
        "type": 0,
        "label": "20m",
        "display": "20m Yagi",
        "steppir": 0,
        "rxonly": 0,
        "pa_ant_number": 1,
        "rotator": 1,
        "rotator_offset": 0,
        "output": { "A1": 1, "A2": 0 }
      }
    }
  }'


  # Modify antenna id 3
curl -X PATCH ... -d '{"serial":"SERIAL","sm":{"obj":{"action":"mod","type":0,"id":3,"label":"40m","display":"40m Dipole"}}}'

  # Remove antenna id 3
curl -X PATCH ... -d '{"serial":"SERIAL","sm":{"obj":{"action":"rem","id":3}}}'
